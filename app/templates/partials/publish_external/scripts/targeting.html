'…' : ''); cnt.textContent = picked.length + ' selected'; cnt.style.display = 'inline-block'; } } // close the dropdown when clicking anywhere else document.addEventListener('click', e => { const dd = document.getElementById('occDD'); if (dd && !dd.contains(e.target)) dd.classList.remove('open'); }); // reflect any preselected occupations (edit flow) in the dropdown label occDDChanged();
