function isInterviewTask(value) { return ['interview', 'online_interview', 'remote_interview', 'in_person', 'in_person_study'].includes(value); } function isOnlineInterviewTask(value) { return ['online_interview', 'remote_interview'].includes(value); } function syncAdminStudyType() { const typeSelect = document.getElementById('taskTypeSelect'); const value = typeSelect?.value || 'survey'; const interview = isInterviewTask(value); const online = isOnlineInterviewTask(value); const inPerson = value === 'in_person' || value === 'in_person_study'; const interviewBox = document.getElementById('adminInterviewFields'); const meetingField = document.getElementById('adminMeetingLinkField'); const locationField = document.getElementById('adminLocationField'); const bookableTimesField = document.getElementById('adminBookableTimesField'); const linkField = document.getElementById('surveyLinkField'); const formUrl = document.getElementById('formUrlInput'); const formUrlLabel = document.getElementById('formUrlLabel'); const formatSelect = document.getElementById('participationFormatSelect'); if (interviewBox) interviewBox.style.display = interview ? 'block' : 'none'; if (meetingField) meetingField.style.display = 'none'; if (locationField) locationField.style.display = inPerson ? 'block' : 'none'; if (bookableTimesField) bookableTimesField.style.display = inPerson ? 'block' : 'none'; if (linkField) linkField.style.display = inPerson ? 'none' : 'block'; if (formUrl) { formUrl.required = !inPerson; if (online) { formUrl.placeholder = 'https://calendly.com/... or https://zoom.us/j/...'; } else { formUrl.placeholder = 'https://your-survey-link.com/...'; } if (inPerson) formUrl.closest('.field')?.classList.remove('error'); } if (formUrlLabel) { formUrlLabel.innerHTML = online ? 'Survey / Calendly / meeting link *' : 'Survey link *'; } if (formatSelect) { if (online) formatSelect.value = 'Video interview'; else if (inPerson) formatSelect.value = 'In-person study'; else if (value === 'survey') formatSelect.value = 'Online surveys'; } updateCommission(); } function slotLabel(slot) { if (!slot?.date || !slot?.start || !slot?.end) return ''; const date = new Date(`${slot.date}T00:00:00`); const dateLabel = Number.isNaN(date.getTime()) ? slot.date : date.toLocaleDateString('en-US', { weekday:'short', month:'short', day:'numeric' }); return `${dateLabel}, ${slot.start} - ${slot.end}`; } function renderAdminSlots() { const list = document.getElementById('adminSlotList'); const hidden = document.getElementById('availabilitySlotsInput'); if (hidden) hidden.value = JSON.stringify(adminSlots); if (!list) return; if (!adminSlots.length) { list.innerHTML = 'No bookable times added yet.'; return; } list.innerHTML = adminSlots.map((slot, index) => `${slot.label}` ).join(''); } function addAdminSlot() { const date = document.getElementById('slotDateInput')?.value; const start = document.getElementById('slotStartInput')?.value; const end = document.getElementById('slotEndInput')?.value; if (!date || !start || !end) return; const slot = { date, start, end, timezone: Intl.DateTimeFormat().resolvedOptions().timeZone }; slot.label = slotLabel(slot); if (!adminSlots.some(existing => existing.date === date && existing.start === start && existing.end === end)) { adminSlots.push(slot); adminSlots.sort((a, b) => `${a.date} ${a.start}`.localeCompare(`${b.date} ${b.start}`)); renderAdminSlots(); } } function removeAdminSlot(index) { adminSlots.splice(index, 1); renderAdminSlots(); }
