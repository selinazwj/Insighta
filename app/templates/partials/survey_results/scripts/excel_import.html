async function uploadQualityExcel(event) { event.preventDefault(); const input = document.getElementById('qualityExcelFile'); const btn = document.getElementById('qualityUploadBtn'); const status = document.getElementById('qualityImportStatus'); if (!input || !input.files || !input.files[0]) return; const data = new FormData(); data.append('file', input.files[0]); data.append('auto_approve', '{{ "1" if survey.quality_auto_filter_enabled else "0" }}'); data.append('auto_approve_min_score', '{{ survey.quality_auto_filter_min_score or 80 }}'); data.append('use_llm', document.getElementById('qualityUseLlm')?.checked ? '1' : '0'); btn.disabled = true; status.className = 'quality-import-status'; status.textContent = ''; try { const res = await fetch(`/api/surveys/${SURVEY_ID}/quality/import-excel`, { method: 'POST', credentials: 'same-origin', body: data }); const payload = await res.json().catch(() => ({})); if (!res.ok) throw new Error(payload.detail || payload.message || 'Upload failed'); status.className = 'quality-import-status success'; status.textContent = payload.queued ? `Import queued. Job #${payload.job_id || ''} will process in the background.` : `Imported ${payload.imported_rows || 0} rows. Quality checks are ready.`; input.value = ''; if (payload.queued) { setTimeout(loadQualityResults, 3000); } else { await loadQualityResults(); } } catch (e) { status.className = 'quality-import-status error'; status.textContent = e.message || 'Upload failed.'; } finally { btn.disabled = false; } }
