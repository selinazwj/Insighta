// ── Render everything ───────────────────────────────────── function renderResults(data) { // Header document.getElementById('surveyTitle').innerHTML = `${escHtml(data.survey_title)} results`; document.getElementById('surveySubtitle').textContent = `${data.total_responses} response${data.total_responses !== 1 ? 's' : ''} collected`; // Stats document.getElementById('statTotal').textContent = data.total_responses; document.getElementById('statQuestions').textContent = data.questions.length; document.getElementById('statCompletion').textContent = data.total_responses > 0 ? '100%' : '—'; document.getElementById('statsRow').style.display = 'grid'; // Questions const list = document.getElementById('resultsList'); if (data.questions.length === 0 || data.total_responses === 0) { list.innerHTML = '

No responses yet.

'; return; } list.innerHTML = data.questions.map((q, i) => renderQResult(q, i)).join(''); // Show AI section document.getElementById('aiSection').style.display = 'block'; } // ── Render a single question result ────────────────────── function renderQResult(q, index) { const typeLabels = { single: 'Single choice', multiple: 'Multiple choice', text: 'Text', scale: 'Scale', dropdown: 'Dropdown' }; let bodyHtml = ''; if (['single', 'multiple', 'dropdown'].includes(q.question_type)) { const dist = q.distribution || {}; const total = Object.values(dist).reduce((a, b) => a + b, 0) || 1; const sorted = Object.entries(dist).sort((a, b) => b[1] - a[1]); bodyHtml = `
${q.total_answers} answer${q.total_answers !== 1 ? 's' : ''}
${sorted.map(([label, count]) => { const pct = Math.round((count / total) * 100); return `
${escHtml(label)}
${pct}%
${count}
`; }).join('')}
`; } else if (q.question_type === 'scale') { const dist = q.distribution || {}; const total = Object.values(dist).reduce((a, b) => a + b, 0) || 1; bodyHtml = `
${q.average ?? '—'}
/ 5 average · ${q.total_answers} answer${q.total_answers !== 1 ? 's' : ''}
${[1,2,3,4,5].map(n => { const count = dist[String(n)] || 0; const pct = Math.round((count / total) * 100); return `
${n}
${pct > 8 ? `${pct}%` : ''}
${count}
`; }).join('')}
`; } else if (q.question_type === 'text') { const responses = q.responses || []; bodyHtml = `
${responses.length} response${responses.length !== 1 ? 's' : ''}
${responses.length === 0 ? '
No responses yet.
' : responses.map(r => `
${escHtml(String(r))}
`).join('')}
`; } return `
Question ${index + 1}
${escHtml(q.question_text)}
${typeLabels[q.question_type] || q.question_type}
${bodyHtml}
`; }
