function renderQualityResultsTable() { const wrap = document.getElementById('qualityList'); if (!wrap) return; if (!qualityRows.length) { wrap.innerHTML = '
No completed responses yet. Scores appear after participants submit.
'; return; } const summary = qualityRows.reduce((acc, row) => { const label = String(row.quality_label || '').toLowerCase(); const action = reviewStatusToAction(row.review_status); if (label === 'high') acc.high += 1; if (label === 'medium' || label === 'low' || label === 'fraud_risk') acc.review += 1; if (action === 'approve') acc.approved += 1; if (action === 'reject') acc.rejected += 1; return acc; }, { high: 0, review: 0, approved: 0, rejected: 0 }); wrap.innerHTML = `
${qualityRows.length}
Total checks
${summary.high}
Clean
${summary.review}
Needs review
${summary.approved}/${summary.rejected}
Approved / rejected
${qualityRows.map(row => renderQualityCard(row)).join('')}
`; } function renderQualityCard(row) { const score = Number(row.quality_score || 0).toFixed(0); const label = row.quality_label || 'medium'; const reasons = (row.reasons || []).map(r => escHtml(String(r))).join(' · ') || 'No triggered reasons.'; return `
${escHtml(row.row_label || '-')}
${escHtml(label)}
${escHtml(qualityDescription(row))}
${reasons}
${score}
Score
${QUALITY_SCOPE === 'builtin' ? `` : ''} ${renderReviewSelect(row)}
`; } function qualityDescription(row) { const label = String(row.quality_label || '').toLowerCase(); const score = Number(row.quality_score || 0); const reasons = Array.isArray(row.reasons) ? row.reasons.filter(Boolean) : []; const triggered = Array.isArray(row.triggered_rules) ? row.triggered_rules.filter(Boolean) : []; if (label === 'fraud_risk') { return 'High-risk response. Review identity, duplicate, device, timing, and answer consistency signals before approving payout.'; } if (label === 'low') { return reasons[0] ? `Low-quality response flagged mainly because: ${String(reasons[0])}` : 'Low-quality response. Please inspect answers before approving payout.'; } if (label === 'medium') { return reasons[0] || triggered[0] ? `Some quality signals need review: ${String(reasons[0] || triggered[0])}` : 'Mixed quality signals. Review if the response looks unusual.'; } if (score >= 80 && !reasons.length && !triggered.length) { return 'No major issues detected. This response looks ready for approval.'; } return reasons[0] || triggered[0] || 'Quality check completed. Open the row details if you need more context.'; } function updateQualityRowInList(checkId, patch) { const idx = qualityRows.findIndex(row => row.id === checkId); if (idx < 0) return; qualityRows[idx] = { ...qualityRows[idx], ...patch }; renderQualityResultsTable(); }
