Loading listings...
'; try { const res = await fetch(`/admin/listings?admin_key=${encodeURIComponent(adminKey)}&q=${encodeURIComponent(q)}`, {cache:'no-store'}); if (!res.ok) throw new Error((await res.json().catch(()=>({}))).detail || 'Failed to load listings'); listings = await res.json(); renderListings(); } catch(err) { wrap.innerHTML = `
${esc(err.message)}
`; } } function renderListings() { const wrap = document.getElementById('listingsResults'); if (!listings.length) { wrap.innerHTML = '
No listings found

${esc(s.title)}
${esc(s.description || '')}
${esc(s.status || '')}
${esc(s.payment_status || '')} ${esc(s.task_type_label || s.task_type || '')}
${esc(s.category || '')} $${Number(displayReward || 0).toFixed(2)} ${s.completed_count || 0}/${s.target_responses || 0} ${esc(s.publisher_email || '')} ${created} `; }).join(''); wrap.innerHTML = `
${rows}
ListingStatusTypeRewardResponsesPublisherCreated
`; } async function deleteListing(id, title) { if (!confirm(`Delete listing "${title}"? This also deletes related responses and quality data.`)) return; try { const res = await fetch(`/admin/listings/${id}/delete`, { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ admin_key: adminKey }) }); if (!res.ok) throw new Error((await res.json().catch(()=>({}))).detail || 'Delete failed'); listings = listings.filter(s => s.id !== id); renderListings(); } catch(err) { alert(err.message); } } function editListing(id) { if (!adminKey) return; window.location.href = `/admin/listings/${id}/edit?admin_key=${encodeURIComponent(adminKey)}`; } async function copyListingLink(url) { if (!url) { alert('This listing does not have a share link yet.'); return; } try { await navigator.clipboard.writeText(url); alert('Share link copied.'); } catch { prompt('Copy this link', url); } } async function deleteKnownTestListings() { const titles = [ 'Campus Life Discovery Survey', 'Help Improve Campus Life for Students Like You' ]; if (!confirm(`Delete these test listings if present?\\n\\n${titles.join('\\n')}`)) return; try { const res = await fetch('/admin/listings/delete-by-title', { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ admin_key: adminKey, titles }) }); const data = await res.json().catch(()=>({})); if (!res.ok) throw new Error(data.detail || 'Delete failed'); alert(`Deleted ${data.deleted?.length || 0} listing(s).`); await loadListings(); } catch(err) { alert(err.message); } }
