${esc(err.message)}
`; if (moreBtn) moreBtn.style.display = 'none'; } finally { activityState.loading = false; } } function populateActivitySurveys(surveys, selectedId) { const select = document.getElementById('activitySurvey'); if (!select) return; const current = selectedId || select.value || ''; select.innerHTML = '' + surveys.map(s => ( `` )).join(''); select.value = current; } function renderActivityPeople(people, surveyId) { const wrap = document.getElementById('activityPeople'); if (!surveyId) { wrap.innerHTML = '

if (meta.button) pieces.push(`Button: ${meta.button}`); if (!pieces.length && meta.referrer) pieces.push(`referrer: ${meta.referrer}`); return pieces.slice(0, 4).join(', ') || '—'; }
