// ── DISCOVERY ── async function runDiscovery() { const btn = document.getElementById('discBtn'); const results = document.getElementById('discoveryResults'); const criteria = { population: document.getElementById('dpopulation').value.trim(), location: document.getElementById('dlocation').value.trim() || null, in_person: document.getElementById('dparticipation').value === 'true', age_min: numOrNull('dageMin'), age_max: numOrNull('dageMax'), sample_size: numOrNull('dsampleSize'), study_topic: document.getElementById('dtopic').value.trim() || null, notes: document.getElementById('dnotes').value.trim() || null, }; if (!criteria.population) { alert('Population is required'); return; } btn.disabled = true; btn.textContent = 'Searching...'; results.innerHTML = `
Searching for compliant channels...
`; try { const res = await fetch('/admin/discovery/find', { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ admin_key: adminKey, criteria }) }); const data = await res.json().catch(() => ({})); if (!res.ok) throw new Error(data.detail || 'Discovery failed'); renderDiscovery(data); } catch(err) { results.innerHTML = `
${esc(err.message)}
`; } finally { btn.disabled = false; btn.textContent = 'Find channels'; } } function numOrNull(id) { const v = document.getElementById(id).value; if (!v) return null; const n = Number(v); return isFinite(n) ? n : null; } function renderDiscovery(data) { const channels = data.channels || []; const warnings = (data.warnings || []).filter(Boolean); const results = document.getElementById('discoveryResults'); if (!channels.length) { results.innerHTML = `
No channels found
${esc(data.source||'unknown')}
${esc(data.summary||'Try broadening the criteria.')}
`; return; } const warnHtml = warnings.map(w => `
${esc(w)}
`).join(''); const rows = channels.map(ch => { const urls = [...new Set([ch.url,ch.contact_url].filter(Boolean))]; const urlHtml = urls.length ? urls.map(u=>`${esc(u)}`).join('') : '—'; const localFit = ch.local_fit || ch.population_fit || ''; const ratingMatch = String(localFit).match(/([1-5])\/5/); const ratingHtml = ratingMatch ? `Fit ${ratingMatch[1]}/5` : ''; const fitText = String(localFit).replace(/[1-5]\/5/g,'').trim(); const notes = [ch.compliance_notes,(ch.evidence||[]).slice(0,2).join(' | ')].filter(Boolean).join(' · '); return ` ${esc(ch.name)} ${esc(ch.channel_type)} ${esc(ch.scale_activity||ch.estimated_reach||'Unknown')} ${ratingHtml}${esc(fitText)} ${esc(ch.compliant_contact||ch.access_method||'—')} ${urlHtml} ${esc(notes)} `; }).join(''); results.innerHTML = `
${channels.length} channels found
${esc(data.source||'unknown')}
${esc(data.summary||data.compliance_rule||'')}
${warnHtml}
${rows}
ChannelTypeScale / ActivityFitContact methodURLNotes
`; }
