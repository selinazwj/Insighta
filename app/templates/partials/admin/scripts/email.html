async function sendRewardReminder() { if (!adminKey) return; const btn = document.getElementById('rewardReminderBtn'); const subjectInput = document.getElementById('bulkEmailSubject'); const messageInput = document.getElementById('bulkEmailMessage'); const subject = (subjectInput?.value || '').trim(); const message = (messageInput?.value || '').trim(); if (!subject || !message) { alert('Please enter both a subject and message.'); return; } const original = btn.textContent; btn.disabled = true; btn.textContent = 'Checking...'; try { const preview = await fetch('/admin/users/bulk-email', { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ admin_key: adminKey, dry_run: true, subject, message }) }); if (!preview.ok) throw new Error((await preview.json().catch(()=>({}))).detail || 'Could not preview recipients'); const data = await preview.json(); const count = data.recipient_count || 0; if (!count) { alert('No users with valid email addresses found.'); return; } if (!confirm(`Queue "${subject}" for ${count} user${count === 1 ? '' : 's'}?`)) return; btn.textContent = 'Queueing...'; const res = await fetch('/admin/users/bulk-email', { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ admin_key: adminKey, dry_run: false, subject, message }) }); const result = await res.json().catch(()=>({})); if (!res.ok) throw new Error(result.detail || 'Could not queue reminder emails'); alert(`Email queued for ${result.queued_count || 0}/${result.recipient_count || count} user${(result.recipient_count || count) === 1 ? '' : 's'}.`); } catch(err) { alert(err.message || 'Email queueing failed.'); } finally { btn.disabled = false; btn.textContent = original; } }
