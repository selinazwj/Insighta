'; } } function updateStats() { const pending = allFeedbacks.filter(f => f.status === 'pending').length; const credited = allFeedbacks.filter(f => f.status === 'credited').length; const amount = allFeedbacks.reduce((s,f) => s + (f.credit_amount || 0), 0); document.getElementById('statPending').textContent = pending; document.getElementById('statCredited').textContent = credited; document.getElementById('statAmount').textContent = '$' + amount.toFixed(2); } function filterFeedback(filter, btn) { currentFilter = filter; document.querySelectorAll('.f-tab').forEach(b => b.classList.remove('active')); btn.classList.add('active'); renderFeedbacks(); } function renderFeedbacks() { const list = document.getElementById('feedbackList'); const CAT = { bug:'Bug', feature_request:'Feature', ux:'UX', general:'General', other:'Other' }; let items = allFeedbacks; if (currentFilter === 'pending') items = items.filter(f => f.status === 'pending'); else if (currentFilter === 'credited') items = items.filter(f => f.status === 'credited'); else if (currentFilter === 'rejected') items = items.filter(f => f.status === 'rejected'); else if (['bug','feature_request','ux'].includes(currentFilter)) items = items.filter(f => f.category === currentFilter); if (!items.length) { list.innerHTML = '
No feedback here
'; return; } list.innerHTML = items.map(f => { const date = new Date(f.created_at).toLocaleDateString('en-US',{month:'short',day:'numeric',year:'numeric'}); const cc = f.status === 'credited' ? 'credited' : f.status === 'rejected' ? 'rejected' : ''; const sp = `s-${f.status}`; const actions = f.status === 'pending' ? ` ` : f.status === 'credited' ? `$${f.credit_amount.toFixed(2)} credited` : `Rejected`; return `
${CAT[f.category]||f.category} ${f.status} ${esc(f.user_email)}
${date}
${esc(f.title)}
${esc(f.content)}
${actions}
`; }).join(''); } async function grantCredit(id) { const input = document.getElementById(`amt-${id}`); const amount = parseFloat(input.value); if (!amount || amount <= 0) { alert('Enter a valid amount'); return; } const btn = input.nextElementSibling; btn.disabled = true; btn.textContent = '...'; try { const res = await fetch(`/admin/feedback/${id}/credit`, { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ admin_key: adminKey, amount }) }); if (!res.ok) throw new Error((await res.json()).detail || 'Error'); const f = allFeedbacks.find(x => x.id === id); if (f) { f.status = 'credited'; f.credit_amount = amount; } updateStats(); renderFeedbacks(); } catch(e) { alert(e.message); btn.disabled = false; btn.textContent = 'Grant credit'; } } async function rejectFeedback(id) { if (!confirm('Reject this feedback?')) return; await fetch(`/admin/feedback/${id}/reject`, { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ admin_key: adminKey }) }); const f = allFeedbacks.find(x => x.id === id); if (f) f.status = 'rejected'; updateStats(); renderFeedbacks(); }
