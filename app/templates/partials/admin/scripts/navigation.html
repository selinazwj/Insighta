Try again.'; error.style.display = 'block'; } } function switchPanel(id) { document.querySelectorAll('.panel').forEach(p => p.classList.remove('active')); document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active')); document.getElementById('panel-' + id).classList.add('active'); document.getElementById('nav-' + id).classList.add('active'); document.getElementById('topbar-title').textContent = PANEL_TITLES[id] || id; currentPanel = id; if (id === 'feedback') loadFeedbacks(); if (id === 'support') loadSupportThreads(); if (id === 'listings') loadListings(); if (id === 'analytics') loadAnalytics(); if (id === 'activity') loadActivity(true); if (id === 'jobs') loadBackgroundJobs(); if (id === 'trust') loadTrustReviews(); if (id === 'blocks') loadBlockedUsers(); if (id === 'overview') { loadFeedbacks(); loadSupportThreads(); } } function refreshCurrent() { if (currentPanel === 'feedback') loadFeedbacks(); else if (currentPanel === 'support') loadSupportThreads(); else if (currentPanel === 'listings') loadListings(); else if (currentPanel === 'analytics') loadAnalytics(); else if (currentPanel === 'activity') loadActivity(true); else if (currentPanel === 'jobs') loadBackgroundJobs(); else if (currentPanel === 'trust') loadTrustReviews(); else if (currentPanel === 'blocks') loadBlockedUsers(); else if (currentPanel === 'overview') { loadFeedbacks(); loadSupportThreads(); } }
