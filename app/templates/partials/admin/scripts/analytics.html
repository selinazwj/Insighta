// ── ANALYTICS ── async function loadAnalytics() { const cards = document.getElementById('analyticsCards'); const funnel = document.getElementById('analyticsFunnel'); const activity = document.getElementById('analyticsActivity'); if (!adminKey || !cards) return; funnel.innerHTML = '
Loading funnel...
'; activity.innerHTML = '
Loading activity...
'; try { const res = await fetch(`/admin/analytics?admin_key=${encodeURIComponent(adminKey)}&days=30`, { cache:'no-store' }); if (!res.ok) throw new Error((await res.json().catch(()=>({}))).detail || 'Failed to load analytics'); analytics = await res.json(); renderAnalytics(); } catch(err) { funnel.innerHTML = `
${esc(err.message)}
`; activity.innerHTML = '
Analytics unavailable.
'; } } function renderAnalytics() { if (!analytics) return; const o = analytics.overview || {}; const cardData = [ [fmtNum(o.total_events), 'Total events'], [fmtNum(o.unique_people), 'Unique users'], [fmtNum(o.listing_views), 'Listing views'], [fmtNum(o.start_clicks), 'Starts'], [fmtNum(o.completed_surveys), 'Completed'], [`${Number(o.start_to_complete_rate || 0).toFixed(1)}%`, 'Start to complete'], ]; document.getElementById('analyticsCards').innerHTML = cardData.map(([num,label]) => `
${esc(num)}
${esc(label)}
`).join(''); renderAnalyticsFunnel(analytics.listing_funnel || []); renderAnalyticsActivity(analytics.recent_activity || []); } function renderAnalyticsFunnel(rows) { const wrap = document.getElementById('analyticsFunnel'); const useful = rows.filter(r => (r.views || r.starts || r.completed)); if (!useful.length) { wrap.innerHTML = '
No funnel data yet
Data will appear after users view or start listings.
'; return; } const body = useful.slice(0, 60).map(r => { const startRate = r.start_rate == null ? '—' : `${Number(r.start_rate).toFixed(1)}%`; const completionRate = r.completion_rate == null ? '—' : `${Number(r.completion_rate).toFixed(1)}%`; return `
${esc(r.title)}
${esc(r.status || '')}
${fmtNum(r.views)} ${fmtNum(r.unique_viewers)} ${fmtNum(r.starts)} ${fmtNum(r.completed)} ${startRate} ${completionRate} `; }).join(''); wrap.innerHTML = `
${body}
StudyViewsUnique viewersStartsCompletedStart rateCompletion rate
`; } function renderAnalyticsActivity(items) { const wrap = document.getElementById('analyticsActivity'); if (!items.length) { wrap.innerHTML = '
No recent activity yet.
'; return; } wrap.innerHTML = items.slice(0, 50).map(item => { const label = activityLabel(item); const when = item.created_at ? new Date(item.created_at).toLocaleString('en-US', { month:'short', day:'numeric', hour:'numeric', minute:'2-digit' }) : ''; const target = item.target_label ? ` · ${esc(item.target_label)}` : ''; return `
${label}${target}
${esc(when)} · ${esc(item.page_path || '')}
`; }).join(''); } function activityLabel(item) { const actor = item.actor_type === 'guest' ? `${esc(item.user || 'Guest visitor')} guest` : `${esc(item.user || 'Logged-in user')}`; const names = { listing_view: 'viewed a listing', listing_viewed: 'viewed a listing', study_impression: 'saw a study', study_card_viewed: 'saw a study', study_click: 'clicked Take study', study_start: 'clicked Start', survey_started: 'clicked Start', study_complete: 'completed a study', survey_completed: 'completed a study', sign_up: 'signed up', signup_completed: 'signed up', login: 'signed in', login_success: 'signed in', profile_update: 'updated profile', study_created: 'created a study', study_published: 'published a study', participant_approved: 'approved a participant', participant_rejected: 'rejected a participant', study_shared: 'shared a study', page_view: 'opened a page', page_viewed: 'opened a page', page_exit: 'left a page', page_hidden: 'left a page', dashboard_opened: 'opened participant dashboard', }; return `${actor} ${names[item.event_name] || esc(item.event_name || 'did something')}`; } function fmtNum(n) { return Number(n || 0).toLocaleString('en-US'); }
