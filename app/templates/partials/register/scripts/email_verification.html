Please try again.'; hint.className = 'code-hint error'; btn.disabled = false; btn.textContent = 'Send Code'; } } // ── Auto verify feedback ────────────────────────────────── function autoVerifyCode(val) { const hint = document.getElementById('codeHint'); if (val.length === 6) { hint.textContent = 'Code entered. Click Continue to proceed.'; hint.className = 'code-hint success'; } else if (val.length > 0) { hint.textContent = ''; hint.className = 'code-hint'; } }
