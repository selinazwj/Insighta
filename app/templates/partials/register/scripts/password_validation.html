// ── Password strength + requirements checklist ──────────── function checkPasswordStrength(val) { const fill = document.getElementById('pwBarFill'); const hint = document.getElementById('pwHint'); const strength = document.getElementById('pwStrength'); const checks = { length: (val || '').length >= 8, upper: /[A-Z]/.test(val || ''), lower: /[a-z]/.test(val || ''), number: /\d/.test(val || ''), special: /[^A-Za-z0-9]/.test(val || ''), }; Object.keys(checks).forEach((key) => { const row = document.querySelector(`.pw-rule[data-rule="${key}"]`); if (row) row.classList.toggle('ok', checks[key]); }); if (!val) { strength.hidden = true; fill.style.width = '0%'; hint.textContent = ''; checkPasswordMatch(); return; } strength.hidden = false; const score = Object.values(checks).filter(Boolean).length; fill.style.width = (score / 5) * 100 + '%'; if (score <= 2) { fill.style.background = '#ef4444'; hint.textContent = 'Weak'; hint.style.color = '#ef4444'; } else if (score <= 3) { fill.style.background = '#f59e0b'; hint.textContent = 'Fair'; hint.style.color = '#f59e0b'; } else if (score === 4) { fill.style.background = '#3b82f6'; hint.textContent = 'Good'; hint.style.color = '#3b82f6'; } else { fill.style.background = '#2a5c39'; hint.textContent = 'Strong'; hint.style.color = '#2a5c39'; } checkPasswordMatch(); } function checkPasswordMatch() { const password = document.getElementById('password').value; const confirm = document.getElementById('confirm'); const hint = document.getElementById('confirmHint'); const value = confirm.value; if (!value) { hint.textContent = ''; hint.className = 'confirm-hint'; confirm.style.borderColor = ''; confirm.style.boxShadow = ''; return; } if (value === password) { hint.textContent = 'Passwords match'; hint.className = 'confirm-hint ok'; confirm.style.borderColor = ''; confirm.style.boxShadow = ''; } else { hint.textContent = 'Passwords do not match'; hint.className = 'confirm-hint error'; confirm.style.borderColor = '#ef4444'; confirm.style.boxShadow = '0 0 0 3px rgba(239,68,68,0.1)'; } }
