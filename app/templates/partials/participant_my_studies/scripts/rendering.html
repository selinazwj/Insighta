${m.h}

${m.p}

${activeTab !== 'completed' ? 'Go to Home ' : ''}
`; return; } el.innerHTML = filtered.map(s => `
${iconSvg(s)}
${escapeHtml(s.title)}
${escapeHtml(s.desc)}
${rewardPill(s)}
${s.time}
${cap(s.category)}
${s.is_live_study && Array.isArray(s.booking_slots) && s.booking_slots.length ? `
${s.booking_slots.join(' · ')}
` : ''} ${s.verification_satisfied === false ? `
Verify to get paid
` : ''} ${statusPill(s)} ${rowActions(s)}
`).join(''); } function updateCounts() { const completed = surveys.filter(s => classify(s) === 'completed').length; const inprogress = surveys.filter(s => classify(s) === 'inprogress').length; const skipped = surveys.filter(s => classify(s) === 'skipped').length; const all = surveys.length; document.getElementById('countCompleted').textContent = completed; document.getElementById('countInProgress').textContent = inprogress; document.getElementById('countSkipped').textContent = skipped; document.getElementById('badge-all').textContent = all; document.getElementById('badge-completed').textContent = completed; document.getElementById('badge-inprogress').textContent = inprogress; document.getElementById('badge-skipped').textContent = skipped; }
