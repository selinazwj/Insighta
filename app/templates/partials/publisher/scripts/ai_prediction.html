async function waitForAiJob(jobId, onDone) { for (let attempt = 0; attempt < 120; attempt++) { const res = await fetch(`/api/ai-jobs/${jobId}`, { credentials: 'same-origin', cache: 'no-store' }); const job = await res.json().catch(() => ({})); if (!res.ok) throw new Error(job.detail || 'Could not read AI job'); if (job.status === 'done') return onDone(job.result || {}); if (job.status === 'failed') throw new Error(job.error || 'AI analysis failed'); await new Promise(resolve => setTimeout(resolve, 3000)); } throw new Error('AI analysis is still queued. Refresh this page in a moment.'); } async function queueForecastJob(card, mode, force=false) { const id = card.dataset.surveyId; const body = card.querySelector('.ai-forecast-body'); const scope = card.querySelector('.ai-scope')?.value || 'quick'; body.textContent = `Queueing ${scope} AI analysis…`; try { const res = await fetch(`/api/surveys/${id}/prediction/jobs`, { method: 'POST', credentials: 'same-origin', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ mode, scope, force, display_limit: 8 }), }); const job = await res.json().catch(() => ({})); if (!res.ok) throw new Error(job.detail || 'AI request could not be queued'); body.textContent = `AI analysis queued (job #${job.job_id}). You can leave this page safely.`; await waitForAiJob(job.job_id, data => { if (mode === 'respondents') { const lines = (data.respondents || []).map((r, i) => `${i+1}. ${Math.round((r.completion_probability || 0)*100)}% · ${r.segment_label || 'Respondent'} · ${(r.top_reasons || [])[0] || ''}`); alert(lines.join('\n') || 'No candidate predictions yet.'); body.textContent = 'Top-candidate AI analysis completed.'; return; } const pct = Math.round((data.completion_probability || 0) * 100); const top = data.segment_label || 'General respondents'; const risk = (data.risk_reasons || [])[0] || 'No major risk detected'; const action = data.recommended_action || 'Keep current settings.'; body.innerHTML = `${pct}% expected completion · ${data.candidate_count || 0} candidates · Top group: ${top}
Risk: ${risk}
Suggested action: ${action}`; }); } catch (error) { body.textContent = error.message || 'AI forecast unavailable.'; } } function loadForecast(id) { const card = document.querySelector(`.ai-forecast-card[data-survey-id="${id}"]`); if (card) queueForecastJob(card, 'summary', false); } function refreshForecast(id) { const card = document.querySelector(`.ai-forecast-card[data-survey-id="${id}"]`); if (card) queueForecastJob(card, 'summary', true); } async function showTopRespondents(id) { const card = document.querySelector(`.ai-forecast-card[data-survey-id="${id}"]`); if (card) queueForecastJob(card, 'respondents', false); } // Intentionally no automatic AI fetch here.
